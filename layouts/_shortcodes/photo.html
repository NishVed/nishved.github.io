{{- /* Responsive, lazy-loaded photo via Hugo's image pipeline. Usage: {{< photo src="travel/split-old-town.jpg" alt="..." caption="..." >}} Emits a webp srcset at three widths; falls back gracefully for formats Hugo cannot process. */}} {{- $src := .Get "src" }} {{- $alt := .Get "alt" | default "" }} {{- $caption := .Get "caption" }} {{- $img := resources.Get (printf "img/%s" $src) }} {{- if $img }} {{- $wide := $img }} {{- $srcset := slice }} {{- if ge $img.Width 640 }} {{- $w640 := ($img.Resize "640x webp q80").RelPermalink }} {{- $srcset = $srcset | append (printf "%s 640w" $w640) }} {{- end }} {{- if ge $img.Width 1000 }} {{- $w1000 := ($img.Resize "1000x webp q80").RelPermalink }} {{- $srcset = $srcset | append (printf "%s 1000w" $w1000) }} {{- end }} {{- $maxw := cond (gt $img.Width 1400) 1400 $img.Width }} {{- $main := $img.Resize (printf "%dx webp q80" $maxw) }} {{- $srcset = $srcset | append (printf "%s %dw" $main.RelPermalink $maxw) }} {{- /* Portrait sources (taller than roughly 10:7) get a tighter height cap. */}} {{- $ratio := div (float $img.Width) (float $img.Height) }} {{- $tall := lt $ratio 0.7 }} {{- else }} {{- warnf "photo shortcode: asset not found: img/%s" $src }} {{- end }}
